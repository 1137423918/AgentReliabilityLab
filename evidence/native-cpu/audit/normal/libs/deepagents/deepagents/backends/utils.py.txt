def to_posix_path(path: str) -> str:\n    """Convert a path to POSIX format by replacing backslashes with forward slashes.\n\n    This function normalizes paths by replacing all backslashes with forward slashes,\n    while preserving the existing traversal and allowed-prefix rejection behavior.\n\n    Args:\n        path: The input path string\n\n    Returns:\n        The path with backslashes replaced by forward slashes\n\n    Examples:\n        >>> to_posix_path("C:\\Users\\project\\file.txt")\n        'C:/Users/project/file.txt'\n        >>> to_posix_path("mixed/sep\\path/with\\backslash")\n        'mixed/sep/path/with/backslash'\n        >>> to_posix_path("\\server\\share\\file")\n        '//server/share/file'\n        >>> to_posix_path("already/posix/path")\n        'already/posix/path'\n        >>> to_posix_path("")\n        ''\n        \n    """\n    return path.replace("\\", "/")
