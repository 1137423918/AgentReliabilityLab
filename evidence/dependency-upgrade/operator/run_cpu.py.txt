def save():

def owned():
    rows = {}
    for p in Path('/proc').glob('[0-9]*'):
        try:
            if p.stat().st_uid != os.getuid(): continue
            s = (p/'stat').read_text().rsplit(')', 1)[1].split()
            rows[int(p.name)] = {'parent': int(s[1]), 'start': int(s[19]), 'state': s[0], 'rss': int(s[21])*os.sysconf('SC_PAGE_SIZE')}
        except (OSError, ValueError): pass
    ids = {p.pid for p in [child, server] if p is not None and p.poll() is None and p.pid in rows}
    ids.update(p for p, start in known.items() if p in rows and rows[p]['start'] == start)
    for _ in range(16):
        old = len(ids); ids.update(p for p, row in rows.items() if row['parent'] in ids)
        if len(ids) == old: break
    ids = {p for p in ids if rows[p]['state'] != 'Z'}
    known.update({p: rows[p]['start'] for p in ids})
    return {p: rows[p] for p in ids}


def stop():
    for sig in [signal.SIGTERM, signal.SIGKILL]:
        for pid, value in owned().items():
            if owned().get(pid, {}).get('start') == value['start']:
                try: os.kill(pid, sig)
                except ProcessLookupError: pass
        for p in [child, server]:
            if p and p.poll() is None:
                try: p.wait(timeout=3)
                except subprocess.TimeoutExpired: pass
        if not owned(): break


def argv(command):
    return [str(CORE), str(OUT/'exec_child.py'), str(os.getpid()), str(parent_start), 'cpu', *[str(x) for x in command]]


def run(name, command, cap=120, cwd=SRC, allowed=(0,)):
    global child
    report['active_phase'] = name; save(); start = time.monotonic(); peak = 0
    with (OUT/(name+'.log')).open('wb') as log:
        child = subprocess.Popen(argv(command), cwd=cwd, env=env, stdout=log, stderr=subprocess.STDOUT, start_new_session=True)
        while child.poll() is None:
            rss = sum(v['rss'] for v in owned().values()); peak = max(peak, rss)
            if time.monotonic()-start > cap or time.monotonic()-began > 2400 or rss > 3*2**30 or memory() < 8*2**30:
                raise RuntimeError('CPU time/memory guard: '+name)
            time.sleep(.2)
        code = child.wait()
    report['phases'].append({'name': name, 'exit_code': code, 'wall_seconds': round(time.monotonic()-start, 3), 'peak_owned_RSS_bytes': peak}); save()
    assert code in allowed, name



def download(url, path, *, expected=None, cap_bytes=268435456):
    path.parent.mkdir(parents=True, exist_ok=True)
    started=time.monotonic(); h=hashlib.sha256(); count=0
    with urllib.request.urlopen(url, timeout=30) as response, path.open('xb') as stream:
        headers={k:response.headers.get(k) for k in ['Content-Length','ETag','Last-Modified','x-goog-generation']}
        while chunk:=response.read(1048576):
            count+=len(chunk);assert count<=cap_bytes and time.monotonic()-started<180
            h.update(chunk);stream.write(chunk)
    actual=h.hexdigest()
    if expected:assert actual==expected
    return {'url':url,'sha256':actual,'bytes':count,'headers':headers,'download_seconds':round(time.monotonic()-started,3)}
