"""Fresh bounded CPU-only analysis, with source/input identities fixed first."""
import argparse
import datetime
import hashlib
import importlib.metadata
import json
import os
from pathlib import Path
import resource
import shutil
import socket
import subprocess
import sys
import tempfile
import time


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument('--input', type=Path, required=True)
    args = parser.parse_args()
    source = args.input
    protocol_path = source/'protocol.json'
    p = json.loads(protocol_path.read_text())
    assert sha(source/'run_controller.py') == p['controller_sha256']
    for rel, expected in p['analysis_source_sha256'].items():
        assert sha(source/rel) == expected
    assert {name: importlib.metadata.version(name) for name in p['versions']} == p['versions']
    mem = int(dict(line.split(':', 1) for line in Path('/proc/meminfo').read_text().splitlines())['MemAvailable'].split()[0])*1024
    disk = shutil.disk_usage('/tmp').free
    assert mem >= 16*1024**3 and disk >= 2*1024**3
    ledger = Path('[PRIVATE_PROGRESS]/generation-progress.json')
    assert sha(ledger) == p['ledger_sha256']
    upstream = Path('[PINNED_UPSTREAM]')
    for rel, expected in p['upstream_source_sha256'].items():
        assert sha(upstream/rel) == expected
    out = Path('[SERVER_OUTPUTS]/tool-schema-stage59-20261123-v1')
    out.mkdir(exist_ok=False)
    work = Path(tempfile.mkdtemp(prefix='agentlab-tool-schema-stage59-'))
    shutil.copytree(source, work/'source')
    source = work/'source'
    (out/'location.json').write_text(json.dumps({'work': str(work), 'output': str(out), 'host': socket.gethostname()})+'\n')
    admission = {'utc': datetime.datetime.now(datetime.timezone.utc).isoformat(), 'available_memory_bytes': mem, 'tmp_free_bytes': disk, 'tmp_filesystem': subprocess.check_output(['stat','-f','-c','%T',str(work)],text=True).strip(), 'python': sys.version.split()[0], 'protocol_sha256': sha(protocol_path)}
    assert admission['tmp_filesystem'] in ('ext2/ext3', 'ext4', 'xfs')
    (out/'admission.json').write_text(json.dumps(admission, indent=2)+'\n')
    env = dict(os.environ, PYTHONDONTWRITEBYTECODE='1', OMP_NUM_THREADS='1', MKL_NUM_THREADS='1', OPENBLAS_NUM_THREADS='1', CUDA_VISIBLE_DEVICES='', PYTHONPATH=str(source))
    def limits():
        os.setsid()
        os.nice(19)
        os.sched_setaffinity(0, sorted(os.sched_getaffinity(0))[:2])
        resource.setrlimit(resource.RLIMIT_AS, (2*1024**3, 2*1024**3))
        resource.setrlimit(resource.RLIMIT_CPU, (90, 90))
        resource.setrlimit(resource.RLIMIT_CORE, (0, 0))
    started = time.monotonic()
    results = []
    try:
        for phase, command in [('controls',[sys.executable,str(source/'benchmarks/test_tool_schema.py')]),('analysis',[sys.executable,str(source/'benchmarks/diagnose_saved_tool_calls.py'),'--source',str(source),'--upstream',str(upstream),'--ledger',str(ledger),'--protocol',str(source/'protocol.json'),'--output',str(out/'analysis')])]:
            # Use Popen so even a deadline leaves an explicit owned-child cleanup.
            with (out/(phase+'.log')).open('w') as log:
                proc = subprocess.Popen(command, cwd=source, stdout=log, stderr=subprocess.STDOUT, env=env, preexec_fn=limits)
                try:
                    code = proc.wait(timeout=120)
                except subprocess.TimeoutExpired:
                    import signal
                    os.killpg(proc.pid, signal.SIGKILL)
                    proc.wait(timeout=10)
                    raise
            results.append({'phase':phase, 'exit_code':code})
            if code:
                raise RuntimeError('Phase failed: '+phase)
        assert sha(ledger) == p['ledger_sha256']
        for rel, expected in p['analysis_source_sha256'].items():
            assert sha(source/rel) == expected
        status = 'completed'
    except BaseException as exc:
        status = 'failed'
        results.append({'error_type':type(exc).__name__, 'message':str(exc)[:300]})
    receipt = {'status':status,'phases':results,'protocol_sha256':sha(protocol_path),'wall_seconds':round(time.monotonic()-started,3),'ledger_after_sha256':sha(ledger),'owned_children_running':0,'new_neural_calls':0,'GPU_used':False}
    (out/'run-report.json').write_text(json.dumps(receipt,indent=2)+'\n')
    print(json.dumps({'output':str(out),'receipt':receipt}),flush=True)
    return 0 if status == 'completed' else 2


if __name__ == '__main__':
    raise SystemExit(main())
